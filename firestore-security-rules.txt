// DISHA Firestore Security Rules - Complete Access Control for All Stakeholders
// Version: 1.0
// Last Updated: 2026-08-02
// Compliance: DPDP Act 2023, GDPR-aligned, NEP 2020

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ========== HELPER FUNCTIONS ==========

    // Check if user is authenticated
    function isAuthenticated() {
      return request.auth != null;
    }

    // Check if user is system admin
    function isSuperAdmin() {
      return isAuthenticated() &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'SuperAdmin';
    }

    // Check if user belongs to school
    function belongsToSchool(schoolId) {
      return isAuthenticated() &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.schoolId == schoolId;
    }

    // Check if user is school admin
    function isSchoolAdmin(schoolId) {
      return isAuthenticated() &&
             belongsToSchool(schoolId) &&
             (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'Admin' ||
              get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'Principal');
    }

    // Check if user is teacher
    function isTeacher(schoolId) {
      return isAuthenticated() &&
             belongsToSchool(schoolId) &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'Teacher';
    }

    // Check if user is parent
    function isParent(schoolId) {
      return isAuthenticated() &&
             belongsToSchool(schoolId) &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'Parent';
    }

    // Check if user is student
    function isStudent(schoolId) {
      return isAuthenticated() &&
             belongsToSchool(schoolId) &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'Student';
    }

    // ========== SCHOOLS COLLECTION ==========
    match /schools/{schoolId} {
      // School admin can read/write their school
      allow read: if isSuperAdmin() || isSchoolAdmin(schoolId);
      allow write: if isSuperAdmin() || isSchoolAdmin(schoolId);

      // Teachers/Parents/Students can read school info (for context)
      allow read: if belongsToSchool(schoolId);

      // Subcollections
      match /assessments/{assessmentId} {
        allow read: if isSuperAdmin() || isSchoolAdmin(schoolId) ||
                       (belongsToSchool(schoolId) && request.auth.uid == resource.data.createdBy);
        allow create: if isSchoolAdmin(schoolId);
        allow update: if isSuperAdmin() || isSchoolAdmin(schoolId) ||
                        (request.auth.uid == resource.data.createdBy && resource.data.status == 'Draft');
        allow delete: if isSuperAdmin() || isSchoolAdmin(schoolId);
      }

      match /staff/{staffId} {
        allow read: if isSuperAdmin() || isSchoolAdmin(schoolId) ||
                       (belongsToSchool(schoolId) && request.auth.uid == staffId);
        allow write: if isSuperAdmin() || isSchoolAdmin(schoolId);
      }

      match /students/{studentId} {
        allow read: if isSuperAdmin() || isSchoolAdmin(schoolId) ||
                       (belongsToSchool(schoolId) && request.auth.uid == studentId);
        allow write: if isSuperAdmin() || isSchoolAdmin(schoolId);
      }

      match /workflows/{workflowId} {
        allow read: if isSuperAdmin() || isSchoolAdmin(schoolId) ||
                       belongsToSchool(schoolId);
        allow write: if isSuperAdmin() || isSchoolAdmin(schoolId);
      }
    }

    // ========== USERS COLLECTION ==========
    match /users/{userId} {
      // Users can read/write their own profile
      allow read, write: if request.auth.uid == userId;

      // School admins can read users in their school
      allow read: if isSuperAdmin() ||
                     (isAuthenticated() &&
                      get(/databases/$(database)/documents/schools/$(get(/databases/$(database)/documents/users/$(request.auth.uid)).data.schoolId)).data.schoolId ==
                      get(/databases/$(database)/documents/schools/$(get(/databases/$(database)/documents/users/$(userId)).data.schoolId)).data.schoolId);
    }

    // ========== STAGE 1: FIRST OPINION ASSESSMENTS ==========
    match /stage1_firstOpinionAssessments/{assessmentId} {
      // School admin: Full access to assessments
      allow read: if isSuperAdmin() ||
                     (belongsToSchool(resource.data.schoolId) && isSchoolAdmin(resource.data.schoolId));

      // Creator can read/update their own draft
      allow read: if request.auth.uid == resource.data.createdBy;
      allow update: if request.auth.uid == resource.data.createdBy &&
                       resource.data.status == 'Draft' &&
                       !hasKeys(request.resource.data, ['status', 'calculations']);

      // School admin can approve
      allow update: if isSchoolAdmin(resource.data.schoolId) &&
                       request.resource.data.status == 'Approved';

      allow create: if belongsToSchool(request.resource.data.schoolId);
      allow delete: if isSuperAdmin() || isSchoolAdmin(resource.data.schoolId);
    }

    // ========== STAGE 2: 14-D ASSESSMENTS ==========
    match /stage2_14dAssessments/{assessmentId} {
      allow read: if isSuperAdmin() ||
                     (belongsToSchool(resource.data.schoolId) && isSchoolAdmin(resource.data.schoolId)) ||
                     request.auth.uid == resource.data.createdBy;

      allow create, update: if belongsToSchool(request.resource.data.schoolId) &&
                               (isSchoolAdmin(request.resource.data.schoolId) ||
                                request.auth.uid == request.resource.data.createdBy);

      allow delete: if isSuperAdmin() || isSchoolAdmin(resource.data.schoolId);
    }

    // ========== STAGE 3: IMPROVEMENT PLANS ==========
    match /stage3_improvementPlans/{planId} {
      // Only school admin and plan creator can read
      allow read: if isSuperAdmin() ||
                     isSchoolAdmin(resource.data.schoolId) ||
                     request.auth.uid == resource.data.createdBy;

      // School admin creates plans
      allow create: if isSchoolAdmin(request.resource.data.schoolId);

      // School admin and creator can update
      allow update: if isSuperAdmin() ||
                       isSchoolAdmin(resource.data.schoolId) ||
                       (request.auth.uid == resource.data.createdBy && resource.data.status == 'Draft');

      // Only school admin can approve/activate
      allow update: if isSchoolAdmin(resource.data.schoolId) &&
                       request.resource.data.status in ['Approved', 'Active'];

      allow delete: if isSuperAdmin();
    }

    // ========== CHALLENGES & DIMENSIONS CATALOGS (Read-Only) ==========
    match /challenges_catalog/{challengeId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin();
    }

    match /dimensions_catalog/{dimensionId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin();
    }

    // ========== WORKFLOW TEMPLATES (Read-Only for Schools) ==========
    match /workflow_templates/{templateId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin();
    }

    // ========== USER STAKEHOLDERS ==========
    match /user_stakeholders/{stakeholderId} {
      // Users can read their own stakeholder record
      allow read: if request.auth.uid == stakeholderId;

      // School admin can read stakeholder records for their school
      allow read: if isSuperAdmin() ||
                     (isAuthenticated() &&
                      belongsToSchool(resource.data.schoolId));

      // School admin can update stakeholder settings
      allow update: if isSchoolAdmin(resource.data.schoolId);

      allow write: if isSuperAdmin();
    }

    // ========== ASSESSMENT RESPONSES (Audit Trail) ==========
    match /assessment_responses/{responseId} {
      // Only school admin and responder can read
      allow read: if isSuperAdmin() ||
                     (isAuthenticated() &&
                      get(/databases/$(database)/documents/schools/$(resource.data.schoolId)).data.schoolId == request.auth.token.schoolId &&
                      (isSchoolAdmin(resource.data.schoolId) ||
                       request.auth.uid == resource.data.responderId));

      // Responder can create responses
      allow create: if belongsToSchool(request.resource.data.schoolId);

      // Only admin can update (for correction only, not data change)
      allow update: if isSuperAdmin() || isSchoolAdmin(resource.data.schoolId);

      allow delete: if isSuperAdmin();
    }

    // ========== AUDIT LOGS (Compliance - DPDP Act) ==========
    match /audit_logs/{logId} {
      // Only system admin can read audit logs (compliance requirement)
      allow read: if isSuperAdmin();

      // System writes only
      allow write: if isSuperAdmin();

      // Users cannot modify audit logs (security requirement)
      allow delete: if false;
    }

    // ========== NOTIFICATIONS ==========
    match /notifications/{notificationId} {
      // Users can read notifications addressed to them
      allow read: if request.auth.uid == resource.data.recipientId || isSuperAdmin();

      // Users can update their own notification status
      allow update: if request.auth.uid == resource.data.recipientId &&
                       only(['status', 'readAt'].hasAll(request.resource.data.diff(resource.data).keys()));

      // System creates notifications
      allow create: if isSuperAdmin();

      allow delete: if isSuperAdmin();
    }

    // ========== REPORTING & DASHBOARDS ==========
    match /reporting/{reportId} {
      // School admin can read reports for their school
      allow read: if isSuperAdmin() ||
                     (belongsToSchool(resource.data.schoolId) &&
                      isSchoolAdmin(resource.data.schoolId));

      // System generates reports
      allow write: if isSuperAdmin();
    }

    // ========== BENCHMARK DATA (Read-Only) ==========
    match /benchmark_data/{benchmarkId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin();
    }

    // ========== SYSTEM CONFIGURATION (Admin Only) ==========
    match /system_configuration/{configId} {
      allow read, write: if isSuperAdmin();
    }

    // ========== 14D ASSESSMENT SURVEYS (Anonymous Stakeholder Responses) ==========
    // Allow unauthenticated stakeholders to submit survey responses via public links

    // Full path match for survey responses subcollection
    match /assessments/{assessmentId}/responses/{responseId} {
      // Allow anyone (authenticated or not) to create survey responses
      allow create: if true;
      // Allow anyone to read responses
      allow read: if true;
    }

    // Allow reading assessment collection data
    match /assessments/{assessmentId} {
      allow read: if true;
    }

    // ========== DEFAULT DENY ALL ==========
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

// ========== HELPER FUNCTION TO CHECK IF KEYS EXIST ==========
// Note: Custom function (add to your rules setup)
function only(keys) {
  return keys.size() == 0 || request.resource.data.diff(resource.data).keys().hasAll(keys);
}

function hasKeys(data, keys) {
  return (keys.size() == 0) || (data.keys().hasAll(keys));
}
