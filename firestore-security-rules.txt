rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ====== HELPER FUNCTIONS ======

    // Firestore-doc-based, not custom-claims-based: this app has never called
    // setCustomUserClaims anywhere (confirmed by repo-wide grep), which made
    // every isAdmin()-gated rule in this file permanently unsatisfiable for
    // every real user, including the app's own admin account - see
    // src/pages/Admin.tsx: its user list only ever showed the caller's own
    // account, and Approve/Revoke on anyone else always failed. Reads the
    // caller's own users/{uid} doc instead (the same doc App.tsx already
    // writes `role: 'admin'` to for the hardcoded admin email).
    function isAdmin() {
      return request.auth != null
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isSchoolAdmin(schoolId) {
      return request.auth != null && request.auth.token.schoolId == schoolId
        && (request.auth.token.role == 'admin' || request.auth.token.role == 'principal');
    }

    function isTeacher(schoolId) {
      return request.auth != null && request.auth.token.schoolId == schoolId
        && request.auth.token.role == 'teacher';
    }

    function isParent(schoolId) {
      return request.auth != null && request.auth.token.schoolId == schoolId
        && request.auth.token.role == 'parent';
    }

    function isStudent(schoolId) {
      return request.auth != null && request.auth.token.schoolId == schoolId
        && request.auth.token.role == 'student';
    }

    function isSchoolStakeholder(schoolId) {
      return isSchoolAdmin(schoolId) || isTeacher(schoolId)
        || isParent(schoolId) || isStudent(schoolId);
    }

    // ====== SCHOOLS COLLECTION ======

    match /schools/{schoolId} {
      allow read: if true;
      // isAdmin()/isSchoolAdmin() below depend on Firebase Auth custom claims
      // that this app never sets anywhere (no setCustomUserClaims call exists
      // in the client or Cloud Functions) - see the CHECKUPS SUBCOLLECTION
      // note further down for the same issue, already fixed there the same
      // way. Also, the field names/types here (schoolName/district/timestamp)
      // never matched what saveSchoolToFirestore() in schoolService.ts
      // actually writes (name/city, createdAt as an ISO string) - so school
      // registration has *always* silently failed to persist to Firestore,
      // on two independent counts, since this rule was introduced.
      allow create: if request.auth != null
        && request.resource.data.name is string
        && request.resource.data.city is string;
      allow update: if request.auth != null;
      allow delete: if request.auth != null;

      // ====== ASSESSMENTS SUBCOLLECTION ======

      match /assessments/{assessmentId} {
        allow read: if true;
        allow create: if isSchoolAdmin(schoolId)
          && request.resource.data.assessmentName is string
          && request.resource.data.status == 'ACTIVE'
          && request.resource.data.createdAt is timestamp;
        allow update: if isSchoolAdmin(schoolId);

        // ====== RESPONSES SUBCOLLECTION ======

        match /responses/{responseId} {
          allow read: if true;
          allow create: if request.auth != null
            && request.resource.data.assessmentId == assessmentId
            && request.resource.data.schoolId == schoolId;
          allow update: if request.auth != null
            && request.resource.data.submittedBy == request.auth.uid
            && request.resource.data.status != 'FINAL';
          allow delete: if isAdmin();
        }
      }

      // ====== CHECKUPS SUBCOLLECTION (STAGE 1) ======

      // NOTE (2026-08-31): isSchoolAdmin()/isAdmin() below depend on Firebase
      // Auth custom claims (request.auth.token.role / .schoolId) that this
      // app never actually sets anywhere - there is no setCustomUserClaims
      // call in the client app or Cloud Functions source. Every real
      // authenticated user (including the demo account and the app's own
      // admin email) therefore always fails those checks, so any rule that
      // required them was permanently unsatisfiable and silently blocked
      // its entire feature (confirmed for /checkups: every First Opinion
      // "Analyze & Get First Opinion" save failed with permission-denied
      // regardless of how correct the submitted data was). Relaxed to
      // "signed in" (request.auth != null), which matches how this app's
      // auth actually works today. If real role-based claims are added
      // later (via a Cloud Function on user create/approve), tighten these
      // back down at that point.
      match /checkups/{checkupId} {
        allow read: if true;
        allow create: if request.auth != null
          && request.resource.data.checkupType == 'FirstOpinion'
          && request.resource.data.status == 'SUBMITTED';
        allow update: if request.auth != null;
        allow delete: if request.auth != null;

        // FIXED (2026-09-01): "if true" meant literally anyone, signed in or
        // not, could overwrite another school's saved report - including
        // both the dishaScore/perceptionGap/realInsights AND its
        // inputsChecksum together, so a forged pair would still pass
        // reportIntegrity.ts's own consistency check (it only proves the
        // stored analysis is internally self-consistent, not that Firestore
        // wasn't tampered with from outside the app). Tightened to
        // "signed in", matching every other write rule in this file after
        // the same custom-claims gap noted above checkups/{checkupId}.
        match /analysis/{analysisId} {
          allow read: if true;
          allow create, update: if request.auth != null;
          allow delete: if request.auth != null;
        }
      }

      // Per-day sequence counters backing each checkup's human-readable
      // FO-YYYYMMDD-NN reference number (checkupService.ts's
      // generateCheckupReferenceId). Written inside a Firestore transaction
      // at submit time, so both read (to see the current count) and write
      // (to increment it) are needed by any signed-in user.
      match /checkupCounters/{dateKey} {
        allow read, write: if request.auth != null;
      }

      // ====== REPORTS SUBCOLLECTION (STAGE 2) ======

      match /reports/{reportId} {
        allow read: if true;
        allow create: if request.auth != null;
        allow update: if request.auth != null;
        allow delete: if request.auth != null;
      }

      // ====== SIMULATIONS SUBCOLLECTION (STAGE 3) ======

      match /simulations/{simulationId} {
        allow read: if true;
        allow create: if request.auth != null
          && request.resource.data.scenario is map
          && request.resource.data.baseline is map
          && request.resource.data.createdAt is timestamp;
        allow update: if request.auth != null;
        allow delete: if request.auth != null;

        match /results/{resultId} {
          allow read: if true;
          allow create, update: if true;
          allow delete: if request.auth != null;
        }
      }

      // ====== AUDIT LOGS SUBCOLLECTION ======

      match /auditLogs/{logId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
        allow update: if false;
        allow delete: if request.auth != null;
      }

    }

    // ====== TOP-LEVEL ASSESSMENTS COLLECTION (14D Multi-User Assessment) ======
    // This is a separate, flat `assessments/{assessmentId}` collection (with
    // subcollections like responses, costRates, objectiveData, simulationScenario),
    // distinct from the nested schools/{schoolId}/assessments subcollection above.
    // It backs the 14D Multi-User Assessment / Events feature, where respondents
    // submit without authentication, so access stays open like it was before.
    match /assessments/{document=**} {
      allow read, write: if true;
    }

    // ====== GLOBAL COLLECTIONS ======

    match /dimensionsCatalog/{dimensionId} {
      allow read: if true;
      allow create, update: if isAdmin();
      allow delete: if isAdmin();
    }

    match /challengesCatalog/{challengeId} {
      allow read: if true;
      allow create, update: if isAdmin();
      allow delete: if isAdmin();
    }

    match /users/{userId} {
      allow read: if request.auth != null
        && (request.auth.uid == userId || isAdmin());
      allow create: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.email is string
        && request.resource.data.role is string;
      allow update: if request.auth != null
        && (request.auth.uid == userId || isAdmin());
      allow delete: if isAdmin();
    }

    match /analytics/{analyticsId} {
      allow read: if isAdmin();
      allow create, update: if request.auth == null;
      allow delete: if isAdmin();
    }

    match /systemSettings/{settingId} {
      allow read: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

  }
}
