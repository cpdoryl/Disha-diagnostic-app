## Schema Overview

### Database Structure

```
firestore (Project: disha-production)
│
└── schools/{schoolId}
    ├── assessments/{assessmentId}
    │   └── [Assessment data from Stage 2]
    │
    └── reverseSimulations/{simulationId}
        ├── goalSetting/
        ├── reverseCalculations/
        ├── feasibilityAnalysis/
        ├── actionMapping/
        ├── resourceAllocation/
        ├── timeline/
        └── metadata/
```

### Design Principles

1. **Denormalization where beneficial** - Store calculated data to avoid re-computation
2. **Subcollections for scalability** - Use subcollections for large arrays
3. **Document-level organization** - Keep related data in same document
4. **Timestamp tracking** - All records track created/updated times
5. **Audit trail** - Track who made changes and when

**Validation:**

```typescript
const validateGoalSetting = (data) => {
  const errors = [];
  
  // Current health must be 0-100
  if (data.currentHealth < 0 || data.currentHealth > 100) {
    errors.push('currentHealth must be between 0-100');
  }
  
  // Target health must be >= current health
  if (data.targetHealth < data.currentHealth) {
    errors.push('targetHealth cannot be less than currentHealth');
  }
  
  // Timeline must be valid
  if (![3, 6, 12, 18, 24].includes(data.timeline)) {
    errors.push('timeline must be 3, 6, 12, 18, or 24 months');
  }
  
  // Budget must be positive
  if (data.budget <= 0) {
    errors.push('budget must be positive');
  }
  
  // All dimensions must be 0-100
  for (const [key, value] of Object.entries(data.currentDimensions)) {
    if (value < 0 || value > 100) {
      errors.push(`${key} must be between 0-100`);
    }
  }
  
  return { valid: errors.length === 0, errors };
};
```

## Security Rules

```firestore
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Schools collection
    match /schools/{schoolId} {
      allow read: if request.auth.uid != null && 
                     get(/databases/$(database)/documents/schools/$(schoolId))
                        .data.members[request.auth.uid] != null;
      allow write: if request.auth.uid != null &&
                      hasRole(schoolId, 'admin');
      
      // Reverse Simulations subcollection
      match /reverseSimulations/{simulationId} {
        allow read: if request.auth.uid != null &&
                       isSchoolMember(schoolId);
        
        allow create: if request.auth.uid != null &&
                        isSchoolMember(schoolId) &&
                        validateGoalSetting(request.resource.data.goalSetting);
        
        allow update: if request.auth.uid != null &&
                        isSchoolMember(schoolId) &&
                        (resource.data.createdBy == request.auth.uid || hasRole(schoolId, 'admin'));
        
        allow delete: if request.auth.uid != null &&
                        hasRole(schoolId, 'admin');
      }
    }
    
    // Helper functions
    function isSchoolMember(schoolId) {
      return get(/databases/$(database)/documents/schools/$(schoolId))
               .data.members[request.auth.uid] != null;
    }
    
    function hasRole(schoolId, role) {
      let userRole = get(/databases/$(database)/documents/schools/$(schoolId))
                        .data.members[request.auth.uid].role;
      return userRole == role;
    }
    
    function validateGoalSetting(goalSetting) {
      return goalSetting.currentHealth >= 0 &&
             goalSetting.currentHealth <= 100 &&
             goalSetting.targetHealth >= 0 &&
             goalSetting.targetHealth <= 100 &&
             goalSetting.targetHealth >= goalSetting.currentHealth &&
             [3, 6, 12, 18, 24].hasAny([goalSetting.timeline]) &&
             goalSetting.budget > 0;
    }
  }
}
```

async function migrateSimulation(oldData, fromVersion, toVersion) {
  let data = oldData;
  
  if (fromVersion < 2) {
    // Migration from v1 to v2
